function advancePostOffice(){
  // Eight ordered Post Office layers: upper = 1,3,5,7 (0,2,4,6),
  // lower = 2,4,6,8 (1,3,5,7). Each completed action moves exactly one cube,
  // always from the left, first down across all four cubes, then back up across
  // all four cubes, repeating.
  const next=[...postCubeSlots].map(Number);
  const upper=[0,2,4,6], lower=[1,3,5,7];
  const occupied=new Set(next);
  if(postCubeDirection==='down'){
    const source=upper.find(slot=>occupied.has(slot));
    if(source!==undefined){
      const i=next.indexOf(source);
      if(i>=0)next[i]=source+1;
    }
    if(lower.every(slot=>next.includes(slot)))postCubeDirection='up';
  }else{
    const source=lower.find(slot=>occupied.has(slot));
    if(source!==undefined){
      const i=next.indexOf(source);
      if(i>=0)next[i]=source-1;
    }
    if(upper.every(slot=>next.includes(slot)))postCubeDirection='down';
  }
  postCubeSlots=next;
  persistPostOfficeState();
}

function resolveCore(p,name,{skipEncounters=false}={}){
  const info=actionInfo[name];if(!info)return true;const type=info.type;
  if(type==='warehouse'){const gain=refillGood(p,info.good);addLog(`${p.name} fills ${GOOD_LABEL[info.good]} by ${gain} to ${cartCapacity(p)}.`);return true}
  if(type==='wainwright')return unlockCartSlot(p);
  if(type==='post'){
    postUncovered().forEach(r=>r.coins?addLira(p,r.coins):addGood(p,r.good,r.n));
    advancePostOffice();
    // Re-render the board immediately so the grey cube visibly moves to its new Post Office layer.
    addLog(`${p.name} takes the 4 uncovered Post Office rewards.`);
    persistPostOfficeState();
    persistPlayerState();
    render({randomize:false});
    return true;
  }
  if(type==='caravan'){
    caravanPendingCards=[];
    for(let i=0;i<2;i++){
      const card=drawBonus();
      if(card){
        p.bonusHand=(p.bonusHand||[]);p.bonusHand.push(card);caravanPendingCards.push(card);
      }
    }
    p.bonusCards=p.bonusHand.length;
    if(caravanPendingCards.length){
      caravanChoice='discard';
      persistPlayerState();
      addLog(`${p.name} draws ${caravanPendingCards.length} Bonus cards at the Caravansary and must discard 1.`);
    }else{
      addLog('No Bonus cards are available at the Caravansary.');
      caravanChoice=null;
      caravanPendingCards=[];
    }
    return false;
  }
  if(type==='fountain')return false;
  if(type==='palace')return resolvePalace(p);
  if(type==='gem')return resolveGemstone(p);
  if(type==='mosqueSmall'||type==='mosqueGreat')return false;
  if(type==='tea'){
    if(!pendingDice){const target=safeInt(document.querySelector('#tea-target')?.value??teaTarget,teaTarget??6);if(target<3||target>12){addLog('Tea House target must be 3–12.');return false}teaTarget=target;const dice=rollDice();pendingDice={type:'tea',target,dice,total:dice[0]+dice[1],title:'Tea House',canModify:(p.bonusHand||[]).length>0};showBoardDiceAnimation(dice,'Tea House');}
    return false;
  }
  if(type==='black'){
    if(!pendingDice){const good=document.querySelector('#black-good')?.value||'fabric';const dice=rollDice();pendingDice={type:'black',extraGood:good,dice,total:dice[0]+dice[1],title:'Black Market',canModify:(p.bonusHand||[]).length>0};showBoardDiceAnimation(dice,'Black Market');}
    return false;
  }
  if(type==='marketSmall'||type==='marketLarge'){
    const market=type==='marketSmall'?'small':'large',idx=currentDemand(market),need=demandCounts(idx);const qs={...marketQuantities};let count=0;
    document.querySelectorAll('.market-q').forEach(i=>{const g=i.dataset.good;qs[g]=Math.max(0,Math.min(need[g]||0,p.goods[g]||0,safeInt(i.value,0)));i.value=String(qs[g]);count+=qs[g]});
    if(count<1||count>5){addLog('Choose between 1 and 5 goods to sell.');return false}
    if(GOODS.some(g=>qs[g]>(p.goods[g]||0))){addLog('You cannot sell more goods than you carry.');return false}
    GOODS.forEach(g=>{p.goods[g]-=qs[g];syncSultanCubeToGood(p,g)});p.coins+=MARKET_PAYOUT[count];advanceDemand(market);marketQuantities=qs;addLog(`${p.name} sells ${count} demanded goods at the ${market==='small'?'Small':'Large'} Market for ${MARKET_PAYOUT[count]} Lira.`);persistPlayerState();return true;
  }
  if(type==='police'){
    if(!p.familyPos||p.familyPos!=='Police Station')return true;
    familyActionTarget=familyActionTarget||'Fountain';familyActionMode=true;p.familyPos=familyActionTarget;beginTileAction(p,familyActionTarget);addLog(`${p.name} frees the Family Member to ${familyActionTarget}.`);return false;
  }
  return true;
}

function finishAction(placeAssistant=false){
  const p=activePlayer();
  if(placeAssistant && pendingAssistantDrop && p && p.pos!=='Fountain' && p.assistants>0 && !p.left.includes(p.pos)){
    p.assistants--;
    p.left.push(p.pos);
    addLog(`${p.name} completes the action at ${p.pos} and leaves an Assistant there.`);
  }
  pendingAssistantDrop=false;
  awaitingAction=false;actionInitiated=true;familyActionMode=false;familyActionTarget=null;teaTarget=null;marketQuantities={fabric:0,spice:0,fruit:0,heirloom:0};caravanChoice=null;caravanPendingCards=[];persistPlayerState();
  if(endGameTrigger()&&!gameOver)addLog('The game is entering the final round after this turn.');
  resolvePendingFamilyRewards();
}

}
